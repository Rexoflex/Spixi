const r2 = (n) => String(Math.round(n * 100) / 100);
// the tail path, scaled from the 8×13 master; the 1px seam overlap stays absolute
function paths(w, h) {
  const sx = w / 8, sy = h / 13, W = w + 1;
  const P = [[2.2, 0], [0, 0.2], [0.5, 2.2], [3.4, 8.2], [8, 13]].map(([x, y]) => [x * sx, y * sy]);
  const recv = `M${W} 0 L${r2(P[0][0])} 0 Q${r2(P[1][0])} ${r2(P[1][1])} ${r2(P[2][0])} ${r2(P[2][1])} Q${r2(P[3][0])} ${r2(P[3][1])} ${r2(P[4][0])} ${r2(P[4][1])} L${W} ${r2(h)} Z`;
  const m = (x) => r2(W - x);
  const sent = `M0 0 L${m(P[0][0])} 0 Q${m(P[1][0])} ${r2(P[1][1])} ${m(P[2][0])} ${r2(P[2][1])} Q${m(P[3][0])} ${r2(P[3][1])} ${m(P[4][0])} ${r2(P[4][1])} L0 ${r2(h)} Z`;
  return { recv, sent };
}
