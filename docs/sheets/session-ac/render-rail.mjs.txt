/* generatePage's carriers, substituted the way C# does it: a temp copy beside the built shell so
   every relative asset resolves; *SL{SpixiPlatform} + the four inset carriers */
function localized(file, platform, insets) {
  let html = readFileSync(join(process.cwd(), root, file), 'utf8');
  html = html.split('*SL{SpixiPlatform}').join(platform || '');
  const ins = insets || { top: 0, bottom: 0, left: 0, right: 0 };
  html = html.split('*SL{AndroidInsetTop}').join(String(ins.top)).split('*SL{AndroidInsetBottom}').join(String(ins.bottom)).split('*SL{AndroidInsetLeft}').join(String(ins.left)).split('*SL{AndroidInsetRight}').join(String(ins.right));
  const out = join(process.cwd(), root, '_rail-' + platform + '-' + file);
  writeFileSync(out, html); temps.push(out); return out;
}
async function boot(ctx, file, platform, insets, device) {
  const page = await ctx.newPage();
  if (device) await page.addInitScript((d) => {
    // the DEVICE, as landscape-runtime.js reads it: screen.width/height + screen.orientation
    const listeners = [];
    const so = { get type() { return d.landscape ? 'landscape-primary' : 'portrait-primary'; }, addEventListener: (t, fn) => listeners.push(fn), removeEventListener: (t, fn) => { const i = listeners.indexOf(fn); if (i >= 0) listeners.splice(i, 1); } };
    Object.defineProperty(window.screen, 'width', { get: () => d.landscape ? d.long : d.short, configurable: true });
    Object.defineProperty(window.screen, 'height', { get: () => d.landscape ? d.short : d.long, configurable: true });
    Object.defineProperty(window.screen, 'orientation', { get: () => so, configurable: true });
    window.__rotate = (landscape) => { d.landscape = landscape; listeners.slice().forEach((fn) => fn()); };
  }, device);
  await page.goto('file://' + localized(file, platform, insets) + '?mobile=1');
  await page.waitForFunction(() => typeof window.executeUiCommand === 'function');
  return page;
}
