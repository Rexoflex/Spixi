async function open(shell, theme) {
  const ctx = await browser.newContext({ viewport: VP, deviceScaleFactor: 2.5, colorScheme: theme, reducedMotion: 'reduce', userAgent: UA });
  const page = await ctx.newPage();
  await page.addInitScript(({ theme }) => { try { localStorage.setItem('spixi.theme', theme); } catch (e) {} }, { theme });
  await page.goto('file://' + join(RAW, shell) + '?mobile=1');
  await page.waitForFunction(() => typeof window.executeUiCommand === 'function');
  await page.evaluate(({ theme }) => { window.__push = (fn, ...a) => window.executeUiCommand(window[fn], ...a.map((v) => btoa(unescape(encodeURIComponent(String(v)))))); window.__push('setTheme', theme); }, { theme });
  return { ctx, page };
}
