import { createRequire } from 'node:module';
import { readFileSync } from 'node:fs';
const require = createRequire('/opt/node22/lib/node_modules/');
const { chromium } = require('playwright');
const photos = JSON.parse(readFileSync('/tmp/claude-0/-home-claude/4e01758d-41dd-5488-96ff-b22a23e39a49/scratchpad/photos.json', 'utf8'));
const browser = await chromium.launch({ executablePath: '/opt/pw-browsers/chromium-1194/chrome-linux/chrome' });
const UA = 'Mozilla/5.0 (Linux; Android 14; moto g84) AppleWebKit/537.36 Chrome/126.0 Mobile Safari/537.36';
const facts = {};
for (const theme of ['light', 'dark']) {
  const ctx = await browser.newContext({ viewport: { width: 412, height: 1400 }, deviceScaleFactor: 2, colorScheme: theme, reducedMotion: 'reduce', userAgent: UA });
  const page = await ctx.newPage();
  await page.goto('file:///home/claude/Spixi/Spixi/Resources/Raw/html/chat.html?mobile=1');
  await page.waitForFunction(() => typeof window.executeUiCommand === 'function');
  await page.waitForTimeout(600);
  await page.evaluate(({ theme, ph }) => {
    const p = (fn, ...a) => window.executeUiCommand(window[fn], ...a.map((v) => btoa(unescape(encodeURIComponent(String(v))))));
    const T = Math.floor(Date.now() / 1000) - 3600;
    p('setTheme', theme);
    p('onChatScreenReady', 'addrPeer');
    p('setChatMode', '0', '0', '', 'False');
    p('setPhotoPreviews', 'True');
    p('clearMessages', 'false');
    p('addThem', 'r1', 'addrPeer', 'Bob', '', 'Two photos, one close to the chat ground:', String(T));
    p('addFile', 'a1', 'addrPeer', 'Bob', '', 'fa1', 'IMG_near_canvas.jpg', String(T + 60), 'False', 'False', 'False', '100', 'True', 'False', 'True');
    p('addFile', 'a2', 'addrPeer', 'Bob', '', 'fa2', 'IMG_green.jpg', String(T + 70), 'False', 'False', 'False', '100', 'True', 'False', 'True');
    p('addMe', 's1', '', 'Me', '', 'And mine:', String(T + 120), 'True', 'True', 'True', 'False', 'False');
    p('addFile', 'b1', '', 'Me', '', 'fb1', 'IMG_mine_near.jpg', String(T + 130), 'True', 'True', 'True', '100', 'True', 'False', 'True');
    p('addFile', 'b2', '', 'Me', '', 'fb2', 'IMG_mine_green.jpg', String(T + 140), 'True', 'True', 'True', '100', 'True', 'False', 'True');
    if (typeof window.messagesDone === 'function') p('messagesDone');
    p('onChatScreenLoaded');
    p('setFileThumb', 'a1', ph[0]); p('setFileThumb', 'a2', ph[1]); p('setFileThumb', 'b1', ph[0]); p('setFileThumb', 'b2', ph[1]);
  }, { theme, ph: photos[theme] });
  await page.waitForTimeout(900);
  facts[theme] = await page.evaluate(() => {
    const o = {};
    for (const id of ['a1', 'b1']) {
      const t = document.querySelector('#messages [data-msgid="' + id + '"] .c-mbubble');
      if (!t) { o[id] = null; continue; }
      const cs = getComputedStyle(t);
      o[id] = { state: t.dataset.state, border: cs.borderTopWidth + ' ' + cs.borderTopColor, shadow: cs.boxShadow };
    }
    const tb = document.querySelector('#messages [data-msgid="r1"] .c-bubble');
    o.textReceived = tb ? { bg: getComputedStyle(tb).backgroundColor, shadow: getComputedStyle(tb).boxShadow } : null;
    return o;
  });
  await page.screenshot({ path: `/home/claude/Spixi/docs/sheets/session4/A5-edge-${theme}.png` });
  await ctx.close();
}
console.log(JSON.stringify(facts, null, 1));
await browser.close();
